    def shutdown(self, force_after_grace: bool = True) -> str:
        """Dua tahap: sopan (terminate + tunggu), lalu tegas (kill)."""
        self.state = RuntimeState.SHUTTING_DOWN
        for c in self.children:
            c.terminate()
        deadline = self._clock() + self.grace_period
        for c in self.children:
            while c.is_alive() and self._clock() < deadline:
                time.sleep(0.05)
        forced = [c.name for c in self.children if c.is_alive()]
        if forced and force_after_grace:
            for c in self.children:
                if c.is_alive():
                    c.kill()
                    self._log("shutdown_forced", c.name)
        else:
            self._log("shutdown_graceful", "semua anak berhenti sopan")
        self.state = RuntimeState.STOPPED
        return "forced" if forced else "graceful"

Gambar 6.1 — Mesin state supervisor + pengalaman sasaran. DEGRADED adalah status RESMI: kegagalan yang disembunyikan lebih
 berbahaya daripada kegagalan yang dilaporkan.
Backoff: kenapa eksponensial, kenapa berbatas
Backoff eksponensial menghormati dua pihak sekaligus: sistem yang barusan crash butuh waktu bernapas
(crash loop instan menggandakan kerusakan), dan pengguna butuh ketenangan (restart setiap 100ms adalah
strobo). Batas 5 percobaan lalu DEGRADED adalah pengakuan jujur: supervisor bukan dewa; setelah lima
kegagalan beruntun, masalahnya bukan keberuntungan — dan laporan terbuka lebih berguna daripada loop
ke-17. Shutdown dua tahap menutup siklus dengan sopan: terminate + grace period (anak bisa menyimpan
state bersih), kill sebagai penghakiman terakhir bagi zombie yang mengabaikan sopan santun.
Kenapa kunci berbasis denyut, bukan PID
Deteksi proses-hidup yang paling sering dipakai komunitas adalah memeriksa PID di berkas kunci — tapi
PID di Windows dipakai ulang OS begitu proses mati: PID 4712 milik Ruka yang mati bisa menjadi PID
4712 milik Chrome lima menit kemudian, dan kunci lama menyatakan 'masuk hidup' untuk proses yang
bahkan bukan Ruka. Deteksi lintas-platform yang jujur justru yang paling sederhana: proses yang hidup
MENULIS sesuatu secara berkala. Denyut jantung tiap 3 detik memberi dua jaminan: instance kedua
menolak jalan dalam 10 detik (timeout), dan instance mati meninggalkan berkas yang basi dalam 10
detik — bisa diambil alih tanpa manual cleanup. Kombinasi PID + boot_token + last_beat dalam satu
berkas JSON memberi diagnosis lengkap: siapa pemegang, kapan denyut terakhir, dan usia kunci.

Anatomi satu siklus monitoring
monitor_once dipanggil scheduler eksternal (loop utama runtime, mis. tiap 3 detik). Alurnya: pukul denyut
(tulis waktu sekarang ke berkas kunci) — cek tiap anak is_alive() — anak mati: naikkan crash_count,
periksa jumlah restart tersisa, catat log crash_detected, jalankan delay backoff, start ulang — kehabisan
percobaan: status DEGRADED + log gave_up (tidak exception! supervisor menelan kegagalan dengan
laporan, bukan menggulingkan seluruh runtime). Sifat 'laporkan, jangan meledak' inilah yang membuat
DEGRADED status resmi: jendela tetap bisa menampilkan status, notifikasi tetap terkirim, dan Bos tetap
tahu. Runtime yang melempar exception di tengah monitoring loop adalah runtime yang mati justru saat
sedang mencoba menyelamatkan diri.

class PermissionPolicy:
    """Peta izin pengguna + penerbitan token konfirmasi.
    grant(tool_name) di-set pengguna (persist di store). Tanpa grant:
    READ boleh, yang lebih tinggi ditolak KECUALI dengan token
    konfirmasi valid untuk parameter PERSIS itu.
    """
    def __init__(self, grants: dict[str, PermissionLevel] | None = None,
                 token_ttl_s: float = 120.0, clock=None):
        self.grants = dict(grants or {})
        self.token_ttl = token_ttl_s
        self._clock = clock or time.time
        self._tokens: dict[str, ConfirmationToken] = {}
    def set_grant(self, tool_name: str, level: PermissionLevel) -> None:
        self.grants[tool_name] = level
    def check(self, spec: ToolSpec, params: dict,
              token: str | None = None) -> tuple[str, str]:
        """Return (putusan, alasan): allowed | needs_confirmation | denied."""
        if spec.permission is PermissionLevel.READ:
            return "allowed", "read: level terendah, tanpa persetujuan"
        granted = self.grants.get(spec.name)
        if granted is not None and LEVEL_ORDER[granted] >= \
                LEVEL_ORDER[spec.permission]:
            return "allowed", f"grant pengguna: {granted.value}"
        if token:
            t = self._tokens.get(token)
            if t is None:
                return "denied", "token konfirmasi tidak dikenal"
            if t.used:
                return "denied", "token sudah dipakai (sekali pakai)"
            if self._clock() > t.expires_at:
                return "denied", "token kedaluwarsa"

    """Registry alat: nama unik, eksekusi terkendali, audit wajib."""
    def __init__(self, policy: PermissionPolicy, audit: AuditLog):
        self.policy = policy
        self.audit = audit
        self._tools: dict[str, ToolSpec] = {}
        self._executors: dict[str, callable] = {}
    def register(self, spec: ToolSpec, executor) -> None:
        if spec.name in self._tools:
            raise ToolError(
                f"alat '{spec.name}' sudah terdaftar — duplikasi "
                "registrasi adalah bug, bukan fitur")
        self._tools[spec.name] = spec
        self._executors[spec.name] = executor

class PathJail:
    """Kurung kerja jalur: resolve + cek prefiks + tolak kabur."""
    def __init__(self, base: str | Path):
        self.base = Path(base).resolve()
        if not self.base.exists():
            raise ToolError(f"base jail tidak ada: {self.base}")
    def confine(self, rel: str) -> Path:
        if not rel:
            raise ToolError("jalur kosong")
        # Pemisah gaya-Windows dinormalisasi di SEMUA platform.
        # Bug asli tertangkap uji keamanan definitif: di Linux,
        # "..\\..\\x" bukan traversal (backslash bagian NAMA),
        # relative_to lolos — jail tampak aman padahal di Windows
        # (target rilis buku!) string yang sama adalah kabur NYATA.
        # Normalisasi dulu, baru resolve + cek prefiks.
        normalized = rel.replace("\\", "/")
        candidate = (self.base / normalized).resolve()
        try:
            candidate.relative_to(self.base)
        except ValueError:
            raise ToolError(
                f"jalur keluar kurung kerja: {rel!r} -> {candidate} "
                f"(base {self.base})") from None
        return candidate

class CommandPolicy:
    """Allowlist-first + denylist + pembatasan bentuk."""
    def check(self, command: str) -> CommandCheck:
        command = command.strip()
        if not command:
            return CommandCheck(False, "perintah kosong", [], "low")
        # shell metacharacter menandakan struktur yang tak kita izinkan
        if re.search(r"[|;&><\n]", command):
            return CommandCheck(
                False, "metacharacter shell (|;&><) tidak diizinkan — "
                "jabarkan satu perintah per langkah", [], "medium")
        argv = command.split()
        if len(argv) > MAX_ARGS:
            return CommandCheck(
                False, f"{len(argv)} argumen > {MAX_ARGS}", [], "medium")
        for pat in DENY_PATTERNS:
            if pat.search(command):
                return CommandCheck(
                    False, f"denylist: cocok pola {pat.pattern!r}",
                    argv, "high")
        for prog, sub in ALLOWED_PREFIXES:
            if argv[0] == prog:
                if not sub:
                    return CommandCheck(True, f"allowlist: {prog}",
                                        argv, "low")
                if len(argv) > 1 and argv[1] in sub:
                    return CommandCheck(
                        True, f"allowlist: {prog} {argv[1]}", argv, "low")
        return CommandCheck(
            False, f"program '{argv[0]}' tidak di allowlist", argv,
            "medium")

"""Protokol IPC lokal — JSONL + token + versi.
Desain (Part XVII):
  - Satu pesan = SATU baris JSON (JSONL). Framing sederhana =
    parsing sederhana = kegagalan yang bisa didiagnosis.
  - Autentikasi: token dibangkitkan runtime Python, ditulis ke
    berkas token (0600 best-effort) di direktori data pengguna.
    Klien (Electron) membaca token itu sebelum menyapa. Tanpa
    token yang benar: server menolak SEMUA method selain hello.
  - Loopback saja: bind 127.0.0.1 — jangan pernah 0.0.0.0. Jaringan
    lokal bukan permukaan serangan yang kita mau buka hari ini.
"""
from __future__ import annotations
import json
import secrets
PROTOCOL_VERSION = 1
MAX_LINE_BYTES = 1 << 20          # 1 MiB: pesan lebih besar ditolak
MAX_QUEUE_PER_METHOD = 256
HELLO_METHOD = "hello"
PING_METHOD = "ping"
class ProtocolError(ValueError):
    """Pelanggaran bentuk protokol."""
def make_token(nbytes: int = 32) -> str:
    return secrets.token_hex(nbytes)
def encode_message(obj: dict) -> bytes:
    """Dict -> baris JSONL (dengan newline)."""
    if not isinstance(obj, dict):
        raise ProtocolError("pesan harus objek")
    line = json.dumps(obj, ensure_ascii=False, separators=(",", ":"))


=== PAGE 109 ===
    if len(line.encode("utf-8")) > MAX_LINE_BYTES:
        raise ProtocolError(
            f"pesan {len(line)} B melebihi {MAX_LINE_BYTES} B — pecah "
            "jadi beberapa pesan, jangan menjejalkan hasil tool mentah")
    return (line + "\n").encode("utf-8")
def decode_message(raw: bytes) -> dict:
    """Baris JSONL -> dict; validasi bidang minimum klien/server."""
    if b"\n" in raw.strip():
        raise ProtocolError("decode_message menerima SATU pesan")
    text = raw.decode("utf-8").strip()
    if not text:
        raise ProtocolError("pesan kosong")
    obj = json.loads(text)
    if not isinstance(obj, dict):
        raise ProtocolError("pesan harus objek")
    if "id" not in obj:
        raise ProtocolError("pesan tanpa 'id' — korelasi tidak mungkin")
    return obj
def make_request(req_id, method: str, params: dict | None = None,
                 token: str = "") -> dict:
    return {"v": PROTOCOL_VERSION, "id": req_id, "method": method,
            "params": params or {}, "token": token}
def make_response(req_id, ok: bool, result=None, error: str = "") -> dict:
    resp = {"v": PROTOCOL_VERSION, "id": req_id, "ok": ok}
    if ok:
        resp["result"] = result
    else:
        resp["error"] = error
    return resp

class IpcClient:
    """Klien dengan reconnect ber-backoff — untuk uji dan untuk


=== PAGE 111 ===
    penggunaan Electron-side (konsep sama di JS)."""
    def __init__(self, endpoint_path: str | Path,
                 max_retries: int = 5, base_delay: float = 0.2):
        self.endpoint_path = Path(endpoint_path)
        self.max_retries = max_retries
        self.base_delay = base_delay
        self._sock: socket.socket | None = None
        self._seq = 0
        self._endpoint: dict | None = None
    def _read_endpoint(self) -> dict:
        data = json.loads(self.endpoint_path.read_text(encoding="utf-8"))
        self._endpoint = data
        return data
    def connect(self) -> None:
        ep = self._read_endpoint()
        last_exc: Exception | None = None
        for attempt in range(self.max_retries):
            try:
                s = socket.create_connection((ep["host"], ep["port"]),
                                             timeout=2.0)
                self._sock = s
                return
            except OSError as exc:
                last_exc = exc
                time.sleep(min(self.base_delay * (2 ** attempt), 5.0))
        raise ConnectionError(
            f"tidak bisa terhubung setelah {self.max_retries} "
            f"percobaan: {last_exc}")
    def call(self, method: str, params: dict | None = None) -> dict:
        if self._sock is None:
            self.connect()
        self._seq += 1
        ep = self._endpoint or self._read_endpoint()
        req = make_request(self._seq, method, params, ep["token"])
        self._sock.sendall(encode_message(req))
        buf = b""
        while b"\n" not in buf:
            chunk = self._sock.recv(65536)
            if not chunk:
                raise ConnectionError("server menutup koneksi")
            buf += chunk
        return decode_message(buf)
    def close(self) -> None:
        if self._sock:
            self._sock.close()
            self._sock = None

Ukuran pesan: mengapa 1 MiB cukup dan 100 MiB bukan
Batas MAX_LINE_BYTES 1 MiB bukan angka modem: pesan perintah (method + params + token) wajar
di bawah 10 KiB; hasil alat yang besar (isi berkas, output panjang) adalah DATA yang harus melewati jalur
chunking atau ditulis ke disk + dikirim sebagai referensi path. Menjejalkan 100 MiB ke satu baris JSON
memperlambat tiga pihak sekaligus: server (parse), klien (memori), dan manusia (diagnosis: baris yang tak
bisa dibaca mata). Protokol yang menolak pesan raksasa memaksa disiplin arsitektur yang benar: hasil
besar disimpan, referensi dikirim. Test test_ukuran_maks mengunci batas itu — dan error message-nya
mengajukan solusinya ('pecah jadi beberapa pesan').

Apa yang dijanjikan WAL — dan apa yang tidak
Eksperimen membuktikan bagian pertama janji: transaksi yang belum commit lenyap UTUH — tidak ada
setengah baris. Bagian kedua janji jarang diucapkan tegas: WAL TIDAK menyelamatkan transaksi yang
belum selesai. Lima ribu baris yang ditulis komitmen demi komitmen tetap aman; gelombang konsolidasi
yang sedang menulis komitmen ke-7 saat listrik mati akan kehilangan komitmen itu — dan seluruh desain
Vol V dibangun di atas penerimaan itu: gelombang konsolidasi BOLEH dijalankan ulang (idempoten dari
sisi hasil: dedup akan menemukan ulang fakta yang sama), backup terjadwal membatasi jendela
kehilangan, dan episode mentah masih ada di events untuk dikonsolidasikan kembali. Ketahanan bukan
berarti tak ada yang hilang — berarti tahu PERSIS apa yang boleh hilang dan bisa membangunnya kembali.
Uji memori yang paling sering dilupakan: masa depan
Suite memori menguji masa LALU (persist, retrieve, lupa). Satu kategori uji yang nyaris selalu absen di
dunia nyata: masa DEPAN — TTL yang harus menyala pada waktunya, half-life yang terus meluruh saat
sistem sibuk, preferensi yang ber status di tengah jalan. Vol V mengujinya dengan waktu SINTETIS:
MemoryLifecycle menerima parameter now, ImportanceScorer menerima age_days — keputusan kecil
desain yang membuat waktu bisa DIREKAYASA dalam test: 90 hari berlalu dalam satu assert. Tanpa itu,
test TTL menunggu 90 hari nyata (atau tidur 90 hari dalam CI yang kemudian di-skip semua orang). Waktu
yang bisa disuntik adalah bug masa depan yang bisa ditangkap hari ini.

20.1 WHY — yang tak terlihat tak bisa dipercaya
Agent yang bekerja diam-diam memaksa penggunanya memilih antara dua keyakinan buta: percaya penuh
atau matikan. Log terstruktur adalah jalan ketiga: kepercayaan yang bisa diperiksa.
Disiplin warisan tiga volume ditegaskan ulang: baris log adalah JSON dengan bidang tetap (ts, level,
component, event) + bidang kontekstual; print() tidak punya tempat di jalur produksi karena tak
berstruktur, tak berwaktu, tak terfilter, dan tak teredaksi. Redaksi menyapu tiga pola (AIza, Bearer,
api_key/password/secret/token) SEBELUM menyentuh disk — dan test membaca berkas fisiknya, bukan
memori objek, karena rahasia yang dilindungi hanya di RAM belum dilindungi (pelajaran Patch 1.1 yang
tak akan kedaluwarsa).

Dari 20 pengamuran, percentile-99 interpolasi linier bisa melaporkan angka di ANTARA dua
pengamatan — nilai yang tak pernah terjadi, lahir dari rumus. Nearest-rank memilih elemen yang ADA;
pada n=20, p99 adalah elemen ke-20 = maksimum. Melaporkan 'p99' untuk 20 sampel adalah mengklaim
pengetahuan tentang ekor distribusi dari data yang tak menjangkaunya — maka MetricsRecorder
mengembalikan None hingga n mencapai 100. Kejujuran statistik kecil semacam ini terasa keras; tapi
bandingkan alternatifnya: dashboard yang angkanya indah namun fiktif, dipercaya orang, menjadi dasar
keputusan arsitektur. Vol III sudah membayar tagihan p95-di-bawah-p50; Vol V menolak membayar
tagihan kedua.

def nearest_rank(sorted_values: list[float], q: float) -> float:
    """Percentile nearest-rank: q=0.95 -> ceil(0.95*n)-1 indeks.
    Interpolasi linear pada n kecil bisa menghasilkan p95 < p50 —
    terjadi nyata dan sudah diceritakan di Vol III. Nearest-rank
    tidak pernah melanggar monotisitas posisi.
    """
    if not sorted_values:
        raise ValueError("data kosong")
    if not 0.0 < q <= 1.0:
        raise ValueError("q harus di (0,1]")
    ordered = sorted(sorted_values)
    rank = max(1, math.ceil(q * len(ordered)))
    return ordered[rank - 1]

Rotasi yang tidak rewel
Rotasi ukuran (4 MiB × 3 generasi) dipilih atas rotasi waktu karena log agent desktop tumbuh dengan
LAJU yang tak terduga (hari penuh alat vs hari sepi) — batas ukuran melindungi disk tanpa peduli tempo.
Penamaan .1.log .2.log membuat 'log kemarin' bisa dibaca manusia tanpa alat. Audit JSONL (Part IX)
sengaja DIPISAH dari log operasional: saksi keputusan dan keterangan operasional punya pembaca yang
berbeda (audit dibaca saat investigasi; log dibaca saat debugging) — mencampur keduanya membuat
keduanya membengkak.
