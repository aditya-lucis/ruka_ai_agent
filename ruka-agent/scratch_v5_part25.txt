DEFAULT_CHAIN: tuple[Migration, ...] = (MIGRATION_2, MIGRATION_3,
                                        MIGRATION_4)
class MigrationRunner:
    """Menjalankan rantai migrasi pada satu basis SQLite.
    current_version dibaca dari _migrations (bukan dari bentuk
    skema) — sumber kebenaran tunggal yang eksplisit. Versi
    tujuan lebih rendah dari versi data = tolak jelas (arah
    turun = down() eksplisit, bukan otomatis).
    """
    def __init__(self, conn: sqlite3.Connection,
                 chain: tuple[Migration, ...] = DEFAULT_CHAIN):
        self.conn = conn
        self.chain = sorted(chain, key=lambda m: m.version)
        versions = [m.version for m in self.chain]
        if versions != sorted(set(versions)):
            raise MigrationError(
                f"rantai migrasi tak monoton: {versions}")
        self._ensure_meta()
    def _ensure_meta(self) -> None:
        self.conn.executescript(
            "CREATE TABLE IF NOT EXISTS _migrations ("
            "version INTEGER PRIMARY KEY, description TEXT NOT NULL,"
            " applied_at INTEGER NOT NULL);")
        self.conn.commit()
    def current_version(self) -> int:
        row = self.conn.execute(
            "SELECT MAX(version) FROM _migrations").fetchone()
        return int(row[0] or 0)

    def pending(self) -> list[Migration]:
        cur = self.current_version()
        return [m for m in self.chain if m.version > cur]
    # ------------------------------------------------------ naik/turun
    def upgrade(self, target: int | None = None,
                dry_run: bool = False) -> dict:
        """Terapkan migrasi berurutan sampai target (default: ujung).
        dry_run MENGEMBALIKAN rencana tanpa menyentuh basis —
        direkam di buku sebagai prosedur pra-rilis.
        """
        goal = target if target is not None else \
            self.chain[-1].version
        if goal < self.current_version():
            raise MigrationError(
                f"target v{goal} < data v{self.current_version()} — "
                "penurunan versi butuh down() eksplisit atau backup")
        plan = [m for m in self.chain
                if self.current_version() < m.version <= goal]
        if dry_run:
            return {"from": self.current_version(), "to": goal,
                    "steps": [(m.version, m.description)
                              for m in plan], "executed": False}
        for m in plan:
            m.run_up(self.conn)
        return {"from": (self.current_version() - len(plan))
                if plan else self.current_version(),
                "to": self.current_version(),
                "steps": [(m.version, m.description) for m in plan],
                "executed": True}
    def downgrade(self, target: int) -> dict:
        """Balikkan migrasi turun ke target — eksplisit + bisa gagal
        jujur pada langkah yang tak reversibel."""
        cur = self.current_version()
        if target >= cur:
            raise MigrationError(
                f"target v{target} >= data v{cur} — tak ada yang "
                "diturunkan")
        steps: list[tuple[int, str]] = []
        for m in reversed(self.chain):
            if target < m.version <= cur:
                m.run_down(self.conn)
                steps.append((m.version, f"rollback: {m.description}"))
        return {"from": cur, "to": self.current_version(),
                "steps": steps, "executed": True}
    def status(self) -> dict:
        cur = self.current_version()
        return {"current_version": cur,
                "latest_available": self.chain[-1].version,
                "pending": [(m.version, m.description)
                            for m in self.pending()],
                "chain": [(m.version, m.description,
                           m.reversible) for m in self.chain]}
