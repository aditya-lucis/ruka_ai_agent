    """Perekam latensi per tahap + event terstruktur."""
    def __init__(self):
        self._stages: dict[str, list[float]] = defaultdict(list)
        self.events: list[dict] = []
    def stage(self, name: str, fn, *args, **kwargs):
        """Bungkus satu tahap: catat durasi + hasil/exception (tanpa
        isi sensitif). Return nilai fn."""
        t0 = time.perf_counter()
        try:
            result = fn(*args, **kwargs)
        except Exception as exc:
            self._stages[name].append(
                (time.perf_counter() - t0) * 1000.0)
            self.events.append({
                "stage": name, "ok": False,
                "error": type(exc).__name__,
            })
            raise
        self._stages[name].append((time.perf_counter() - t0) * 1000.0)
        return result
    def record_stage(self, name: str, ms: float):
        """Catat latensi tahap dari luar (adapter async dsb.)."""
        self._stages[name].append(float(ms))
    def media_event(self, stage: str, mime: str, size: int,
                    fingerprint: str, trace_id: str, ok: bool = True):
        """Event media: tanpa bytes, tanpa transkrip."""
        self.events.append({
            "stage": stage, "mime": mime, "bytes": size,
            "fingerprint": fingerprint, "trace": trace_id, "ok": ok,
        })
    def state_transition(self, prev: str, next_: str, cause: str):
        """Event transisi ekspresi (label + sebab, bukan vektor)."""
        self.events.append({
            "stage": "expression", "from": prev, "to": next_,
            "cause": cause,
        })

    def _nearest_rank(sorted_vals: list[float], q: float) -> float:
        """Persentil nearest-rank (Vol III): p = ceil(q/100 * N) elemen
        ke-p pada data terurut — konsisten & anti floor-bias."""
        import math
        n = len(sorted_vals)
        if n == 0:
            return 0.0

        k = max(1, math.ceil(q / 100.0 * n))
        return sorted_vals[min(k, n) - 1]
    def summary(self) -> dict:
        """Ringkasan p50/p95/p99 per tahap + jumlah event."""
        out = {}
        for stage, vals in sorted(self._stages.items()):
            sv = sorted(vals)
            out[stage] = {
                "n": len(sv),
                "p50_ms": round(self._nearest_rank(sv, 50), 3),
                "p95_ms": round(self._nearest_rank(sv, 95), 3),
                "p99_ms": round(self._nearest_rank(sv, 99), 3),
            }
        out["events"] = len(self.events)
        return out

def redact(text: str) -> str:
    """Redaksi secret dalam teks bebas. Konsisten dengan Vol III."""
    out = text
    for pat in _SECRET_PATTERNS:
        out = pat.sub("[REDACTED]", out)
    # tangani nilai setelah label (label=VALUE) yang lolos pola pertama
    out = re.sub(r"(?i)(api[_-]?key|token|secret|password)"
                 r"(\s*[=:]\s*)\S+", r"\1\2[REDACTED]", out)
    return out
def content_fingerprint(data: bytes) -> str:

    menyimpan isi. Ko-lusi kecil dimaklumi dan disebut di buku."""

15.5 PRODUCTION — checklist terakhir volume
Dua puluh butir gerbang rilis — diturunkan langsung dari spesifikasi dan isi Volume IV.
– Semua secret di .env; .env.example lengkap tanpa nilai nyata; .gitignore menutup .env (warisan Patch).
– Validasi input aktif di SEMUA pintu (upload, tool, stream) — bukan cuma jalur utama.
– Batas ukuran & durasi ditegakkan sebelum memori dialokasikan (fail fast, bukan OOM).
– Kuota token gambar dihitung sebelum kirim (listing estimasi; batch besar dipecah).
– VAD terkalibrasi pada UCAPAN NYATA target user; jeda napas tidak memotong kalimat.
– Transkrip berisiko meminta konfirmasi sebelum aksi (verifikasi intent).
– Threshold speaker ditetapkan dari tabel risiko FAR/FRR — bukan default anggap-enak.
– Aksi sensitif menuntut MFA; suara tidak pernah faktor tunggal.
– TTS: verbatim guard aktif untuk kelas pesan faktal; tag audio tidak dipakai pada pesan darurat.
– Expression state tersimpan per sesi terisolasi; transisi terekam label+sebab.
– Avatar fallback neutral teruji; identitas referensi terkunci di review aset.
– Memori: consent gate, TTL sweep terjadwal, forget idempoten, panel user berfungsi.
– Proactive: level izin eksplisit per user; frekuensi nudge dibatasi jendela waktu; tombol mati total.
– Observability: p95 per tahap, redaksi aktif, media tercatat tanpa bytes, trace id lintas lapisan.
– Test suite hijau di CI (243 + integrasi); gradient check tetap menyala.
– Evaluasi golden set: versi model tercatat; regresi kualitas terdeteksi sebelum user mengeluh.
– Integration point Vol III menyala: perception → vector/context/decision/self-model terkontrak bentuk
data.
– Rollback: tiap modul persepsi bisa dimatikan terpisah (feature flag) tanpa menjatuhkan jalur teks.
– Dokumen batas klaim terpasang: tidak ada klaim emosi biologis/kesadaran di UI atau materi produk.
– Kejadian gagal diperlakukan sebagai materi: post-mortem menambah test + baris buku — tradisi empat
volume.

15.9 FUZZING & NEGATIVE TESTING — mencari kehancuran sebelum
user
Invarian menguji jalur benar; fuzzing mencari jalur rusak. Untuk lapisan persepsi, fuzzing murah dan
memalukan bila dilupakan.
– File acak: 100 blob byte acak (dengan magic palsu, terpotong, nol byte, 16 MB+1) → semua harus
ditolak BERSIH dengan PerceptionValidationError — bukan traceback.
– Klaim silang: PNG berisi JPEG, WAV berisi MP3 — 12 kombinasi klaim/nyata harus tertangkap
sniff mismatch.
– Audio ekstrem: hening murni, white noise penuh, DC offset, clipping 100%, 1 sampel, 9,5 jam
penuh — VAD dan framing wajib tetap waras (bug nyata ketiga buku ini hidup di sini).
– State agresif: 1.000 event beruntun, dua event kontradiktif per langkah, NaN dan inf di vektor
context — clamp harus bertahan, tak ada NaN lolos ke label avatar.
– Budget fuzz: jalankan di CI tiap rilis (30 detik); crash baru = test baru + baris failure modes baru —
korpus fuzz TUMBUH dari kecelakaan.
