class AvatarMapper:
    """Mapper L0/L1: label + intensitas -> aset statis & hint blend.
    aset() mengembalikan PATH KONSEPTUAL (kontrak), bukan file nyata —
    pasangan path/aset di-bind oleh aplikasi lewat asset_dir. Ini
    menjaga paket tetap murni logic dan bisa diuji tanpa file gambar.
    """
    def __init__(self, asset_dir: str = "assets/avatar",
                 blend_threshold: float = 0.35):
        if blend_threshold <= 0 or blend_threshold >= 1:
            raise ValueError("blend_threshold di (0,1)")
        self.asset_dir = asset_dir.rstrip("/")
        self.blend_threshold = blend_threshold
    def asset_path(self, label: str) -> str:
        """Path aset untuk label. Label tak dikenal -> neutral
        (fallback eksplisit, bukan exception: avatar rusak tidak
        boleh menjatuhkan seluruh interaksi)."""
        if label not in AVATAR_LABELS and label not in EXTENDED_LABELS:
            label = "neutral"
        return f"{self.asset_dir}/{label}.png"

class AdmissionPolicy:
    """Gerbang evaluation: kandidat layak disimpan atau tidak?
    Rule-based BY DESIGN (Vol III Part VI: rule-based dulu, neural
    hanya bila menang jujur). Skor = bobot importance + keunikan
    ringkasan + tanda eksplisit user ("ingat ini").
    """
    def __init__(self, threshold: float = 0.45):
        if not (0 <= threshold <= 1):
            raise ValueError("threshold di [0,1]")
        self.threshold = threshold
    def evaluate(self, cand: MemoryCandidate) -> tuple[bool, float]:
        explicit = 1.0 if cand.metadata.get("user_flagged") else 0.0
        uniqueness = 1.0 if cand.metadata.get("novel") else 0.3
        score = (0.5 * cand.importance + 0.25 * uniqueness
                 + 0.25 * explicit)
        admitted = score >= self.threshold or explicit > 0
        return admitted, float(score)

14.1 WHY — kontinuitas adalah fitur, keacakan adalah cacat
Spesifikasi Volume IV memberi contoh kegagalan yang tepat: ekspresi happy → extremely angry →
embarrassed → calm pada empat pesan berturut-turut TANPA sebab kontekstual. Pengguna membaca pola
ini sebagai 'rusak' atau ' palsu' — dua vonis yang sama-sama mematikan companion. Sebaliknya,
konsistensi ber-sebab adalah sinyal terpenting kehadiran yang sehat: reaksi wajar atas event, pulih perlahan
setelahnya, dan TIDAK berubah identitas antar hari.
– Consistent identity: persona, gaya, dan referensi avatar stabil lintas sesi (warisan identity Vol II +
asset Part XIII).
– Persistent interaction: sesi baru menyambung kondisi lama dengan sebab yang jelas (user_return
event), bukan reset total.
– Recognition of context: memori & context engine menyambung topik lintas sesi (Vol III).
– Expression continuity: transisi ekspresi divalidasi — PART ini membangun validasinya.
– Interaction continuity: pola respon konsisten (evaluator Part XV mengukurnya).
14.2 IMPLEMENTATION — validator transisi ber-sebab

transisi ekspresi dinilai dari jarak kosakata + dampak event terdekat.
